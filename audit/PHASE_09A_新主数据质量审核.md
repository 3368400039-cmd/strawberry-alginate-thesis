### 4.1 1458公式及缓存的实际性质

## 附录：从新主源复算主要统计的可执行最小代码

以下仅以原件数值列为输入，输出审计用统计，不读旧S1/旧JSON、不写Excel。未包含公式意图解析器，不能冒称真实Excel引擎。全部主表统计可用此代码复核；本报告已列原始异常单元格、方法、双口径和范围，不依赖/tmp长期存在。

```python
from pathlib import Path
from collections import defaultdict
from statistics import mean, stdev
import hashlib, openpyxl
path = Path("实验记录.xlsx")
assert hashlib.sha256(path.read_bytes()).hexdigest() == "a63c8b0e1a07724e861d4b9a379ab632b3c0d41038f77310db36fb3b6b9f66f6"
w = openpyxl.load_workbook(path, read_only=True, data_only=True)
def rows(name):
    return [r for r in w[name].iter_rows(min_row=5, values_only=True) if r[0] is not None]
units = {r[1]: r for r in rows("04_实验单位")}
fruit = {r[2]: r for r in rows("05_果实清单")}
summary = defaultdict(list)
for r in rows("07_连续称重"):
    if r[8] is not None:
        pct = 100 * (r[7] - r[8]) / r[7]
        summary[("mass_loss_full", r[5], r[2])].append(pct)
        summary[("mass_loss_book_rounded", r[5], r[2])].append(round(pct, 2))
unit_decay = defaultdict(int)
for r in rows("08_腐烂观察"):
    if r[7] == "L":
        unit_decay[(r[4], r[2])] += r[8]
for (uid, day), count in unit_decay.items():
    summary[("decay_pct", units[uid][2], day)].append(100 * count / units[uid][4])
fruit_reads = defaultdict(list)
for r in rows("09_品质检测"):
    if r[8] is not None:
        fruit_reads[(r[5], r[2], r[7])].append(r[8])
unit_reads = defaultdict(list)
for (fid, day, metric), reads in fruit_reads.items():
    unit_reads[(fruit[fid][1], day, metric)].append(mean(reads))
for (uid, day, metric), fruit_means in unit_reads.items():
    summary[(metric, units[uid][2], day)].append(mean(fruit_means))
for key, values in sorted(summary.items()):
    print(key, "container_n", len(values), "mean", mean(values),
          "sample_SD", stdev(values), "range", min(values), max(values))
w.close()
```
